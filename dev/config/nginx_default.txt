# Which client chunks may be requested, and how a pre-compressed copy of one is chosen.
#
# The app is split into an entry bundle and the dependency chunks beside it (see
# webpack.config.client.js), so the bundle routes below match a name pattern rather than one path.
# The pattern is also what keeps a request from reaching outside dist/client.
#
# The build leaves a .gz and a .br copy beside every file Nginx serves off disk (see
# dev/scripts/compressBundles.js). `gzip_static` picks up the .gz on its own. Brotli has no module
# in the Nginx packaged for Ubuntu 22.04, so it is negotiated by hand: $br_ok says the client
# accepts it, and the request is rewritten into an internal location that serves the .br file with
# the matching Content-Encoding. That location must switch gzip off — left on, it appends its own
# encoding and the browser is told the body is "br, gzip" when it is only brotli.
map $http_accept_encoding $br_ok {
    default     0;
    "~*\bbr\b"  1;
}

# Live server
server {
    listen 80;
    listen [::]:80;
    server_name app.thingspool.net;
    return 301 https://app.thingspool.net/$request_uri;
}
server {
    listen 443 ssl default_server;
    listen [::]:443 ssl default_server;
    server_name app.thingspool.net;
    ssl_certificate /etc/letsencrypt/live/app.thingspool.net/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/app.thingspool.net/privkey.pem;
    root /root/actions-runner/_work/youngjin/youngjin/public;

    # Rate limiting — allow short bursts (up to 10 queued) but enforce 5 req/s average per IP.
    # Returns 429 Too Many Requests when exceeded.
    limit_req zone=per_ip burst=10 nodelay;
    limit_req_status 429;

    # Client bundles — served directly by Nginx (not proxied to Node.js).
    # Live uses promoted copies (*.live.js / style.live.css) mapped to the canonical paths, and
    # falls back to the staging file for a chunk it has never had promoted to it — without that, a
    # chunk introduced by a deploy is missing from the live page until someone promotes.
    # no-cache = browser caches the file but always revalidates via ETag before using it.
    # If unchanged: 304 Not Modified (~200 bytes). If changed: full re-download.
    location ~ ^/app/(?<chunk>bundle|vendor\.[a-z]+)\.js$ {
        root /root/actions-runner/_work/youngjin/youngjin/dist/client;
        default_type application/javascript;
        add_header Cache-Control "no-cache";
        gzip_static on;
        if ($br_ok) { rewrite ^ /__br/$chunk.js last; }
        try_files /$chunk.live.js /$chunk.js =404;
    }
    location ~ ^/__br/(?<chunk>bundle|vendor\.[a-z]+)\.js$ {
        internal;
        root /root/actions-runner/_work/youngjin/youngjin/dist/client;
        default_type application/javascript;
        gzip off;
        gzip_static off;
        add_header Content-Encoding br;
        add_header Cache-Control "no-cache";
        add_header Vary "Accept-Encoding";
        # Only this server's own promoted brotli copy. Anything else hands back to @plain, so a
        # missing .br degrades to gzip rather than serving a different build than the plain route
        # would have.
        try_files /$chunk.live.js.br @plain;
    }
    location @plain {
        root /root/actions-runner/_work/youngjin/youngjin/dist/client;
        default_type application/javascript;
        add_header Cache-Control "no-cache";
        gzip_static on;
        try_files /$chunk.live.js /$chunk.js =404;
    }
    location = /app/style.css {
        alias /root/actions-runner/_work/youngjin/youngjin/dist/client/style.live.css;
        types { text/css css; }
        add_header Cache-Control "no-cache";
        gzip_static on;
    }

    # Served in place of any page whose Node.js process is unreachable — during a deployment, a
    # restart, or a crash. Without it the visitor gets Nginx's bare "502 Bad Gateway", which reads
    # as a broken site rather than as a temporary update.
    # No proxy_intercept_errors here on purpose: these statuses are the ones Nginx itself produces
    # when it cannot reach the upstream, so the app's own error responses still pass through
    # untouched (the health route in particular must keep reporting its real status).
    error_page 502 503 504 /error/deploying.html;
    location = /error/deploying.html {
        root /root/actions-runner/_work/youngjin/youngjin/public;
        internal;
        # "always" is required: the page is delivered with the original 5xx status, and a plain
        # add_header is dropped on anything outside Nginx's default 2xx/3xx set. Without it the
        # browser could cache this page against the URL the visitor was actually trying to reach.
        add_header Cache-Control "no-store" always;
    }

    location / {
        proxy_pass http://localhost:3000;
        proxy_redirect off;

        # Standard proxy headers
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header Host $http_host;
        proxy_set_header X-Forwarded-Proto $scheme;

        # WebSocket support (required for Socket.IO)
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";

        # Long timeout for persistent WebSocket connections
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
    }
}

    location / {
        proxy_pass http://localhost:3001;
        proxy_redirect off;

        # Standard proxy headers

        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header Host $http_host;
        proxy_set_header X-Forwarded-Proto $scheme;
